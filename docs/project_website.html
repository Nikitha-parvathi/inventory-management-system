Inventory Management System

MySQL Individual Project – Project 8

Problem Statement

A retail business must track products, suppliers, purchases, sales and available stock across warehouses. Without a central database, stock-outs and over-stocking go unnoticed. This project builds a MySQL database that monitors inventory in real time and raises low-stock alerts.

Objectives

Centralise product and supplier data · record purchases and sales · keep stock accurate per warehouse · alert when stock falls to the reorder level · produce purchase, sales and movement reports.

Modules

1 Suppliers – vendor details · 2 Product Categories · 3 Products – SKU, prices, reorder level · 4 Warehouses (plus a stock table for quantity per warehouse) · 5 Purchase Orders · 6 Purchase Items · 7 Sales · 8 Sales Items · 9 Stock Transactions – IN/OUT audit log.

Functional Requirements

Add/update suppliers, categories, products · raise purchase orders and receive goods (stock increases) · record sales (stock decreases, blocked if insufficient) · view current stock · low-stock alerts · supplier-wise purchases · monthly sales · fast/slow movers.

How the System Works

A product belongs to one category and one supplier. A purchase order lists products in purchase_items; when received, a trigger/procedure adds quantity to stock and logs an IN row in stock_transactions. A sale lists products in sales_items; the procedure sp_make_sale reduces stock and logs an OUT row. Views compare stock with reorder_level to produce alerts and reports.

ER Diagram

PK = primary key, FK = foreign key. Many-to-many between products and purchase orders / sales is resolved by purchase_items and sales_items. All links are One-to-Many (1:N).

Database Schema

CREATE DATABASE inventory_db; USE inventory_db;

CREATE TABLE suppliers(
 supplier_id INT AUTO_INCREMENT PRIMARY KEY,
 name VARCHAR(80) NOT NULL, email VARCHAR(80) UNIQUE,
 phone VARCHAR(15), city VARCHAR(40),
 status VARCHAR(10) NOT NULL DEFAULT 'Active' CHECK (status IN ('Active','Inactive')));

CREATE TABLE categories(
 category_id INT AUTO_INCREMENT PRIMARY KEY,
 name VARCHAR(50) NOT NULL UNIQUE, description VARCHAR(120));

CREATE TABLE products(
 product_id INT AUTO_INCREMENT PRIMARY KEY,
 sku VARCHAR(15) NOT NULL UNIQUE, name VARCHAR(80) NOT NULL,
 category_id INT NOT NULL, supplier_id INT NOT NULL,
 unit_price DECIMAL(10,2) NOT NULL CHECK (unit_price > 0),
 cost_price DECIMAL(10,2) NOT NULL CHECK (cost_price >= 0),
 reorder_level INT NOT NULL DEFAULT 10,
 FOREIGN KEY (category_id) REFERENCES categories(category_id),
 FOREIGN KEY (supplier_id) REFERENCES suppliers(supplier_id));

CREATE TABLE warehouses(
 warehouse_id INT AUTO_INCREMENT PRIMARY KEY,
 name VARCHAR(50) NOT NULL, city VARCHAR(40) NOT NULL);

CREATE TABLE stock(
 product_id INT, warehouse_id INT,
 quantity INT NOT NULL DEFAULT 0 CHECK (quantity >= 0),
 PRIMARY KEY (product_id, warehouse_id),
 FOREIGN KEY (product_id) REFERENCES products(product_id),
 FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id));

CREATE TABLE purchase_orders(
 po_id INT AUTO_INCREMENT PRIMARY KEY, supplier_id INT NOT NULL,
 order_date DATE NOT NULL,
 status VARCHAR(10) NOT NULL DEFAULT 'Pending' CHECK (status IN ('Pending','Received','Cancelled')),
 FOREIGN KEY (supplier_id) REFERENCES suppliers(supplier_id));

CREATE TABLE purchase_items(
 item_id INT AUTO_INCREMENT PRIMARY KEY, po_id INT NOT NULL, product_id INT NOT NULL,
 quantity INT NOT NULL CHECK (quantity > 0), unit_cost DECIMAL(10,2) NOT NULL,
 FOREIGN KEY (po_id) REFERENCES purchase_orders(po_id),
 FOREIGN KEY (product_id) REFERENCES products(product_id));

CREATE TABLE sales(
 sale_id INT AUTO_INCREMENT PRIMARY KEY, customer_name VARCHAR(60) NOT NULL,
 sale_date DATE NOT NULL, warehouse_id INT NOT NULL,
 payment_status VARCHAR(10) NOT NULL DEFAULT 'Pending' CHECK (payment_status IN ('Paid','Pending')),
 FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id));

CREATE TABLE sales_items(
 item_id INT AUTO_INCREMENT PRIMARY KEY, sale_id INT NOT NULL, product_id INT NOT NULL,
 quantity INT NOT NULL CHECK (quantity > 0), unit_price DECIMAL(10,2) NOT NULL,
 FOREIGN KEY (sale_id) REFERENCES sales(sale_id),
 FOREIGN KEY (product_id) REFERENCES products(product_id));

CREATE TABLE stock_transactions(
 txn_id INT AUTO_INCREMENT PRIMARY KEY, product_id INT NOT NULL, warehouse_id INT NOT NULL,
 txn_type VARCHAR(3) NOT NULL CHECK (txn_type IN ('IN','OUT')),
 quantity INT NOT NULL CHECK (quantity > 0),
 txn_date DATE NOT NULL, reference VARCHAR(30),
 FOREIGN KEY (product_id) REFERENCES products(product_id),
 FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id));

SQL Queries (10)

1. SELECT + WHERE + ORDER BY – products above ₹1000

SELECT name, unit_price FROM products WHERE unit_price > 1000 ORDER BY unit_price DESC;

2. DISTINCT – cities that supply us

SELECT DISTINCT city FROM suppliers;

3. INSERT – add a new product

INSERT INTO products(sku,name,category_id,supplier_id,unit_price,cost_price,reorder_level)
VALUES ('EL-004','USB Cable',1,1,199,110,25);

4. UPDATE – 5% price rise on stationery

UPDATE products SET unit_price = unit_price*1.05 WHERE category_id = 3;

5. DELETE – remove cancelled purchase orders

DELETE FROM purchase_orders WHERE status='Cancelled';

6. Aggregate – total sales value

SELECT COUNT(DISTINCT sale_id) AS bills, SUM(quantity*unit_price) AS revenue FROM sales_items;

7. GROUP BY + HAVING – products sold more than 20 units

SELECT product_id, SUM(quantity) AS units FROM sales_items GROUP BY product_id HAVING units > 20;

8. String functions – SKU prefix and upper-case names

SELECT UPPER(name) AS product, LEFT(sku,2) AS prefix, CONCAT(sku,' - ',name) AS label FROM products;

9. Date functions – sales per month

SELECT DATE_FORMAT(sale_date,'%Y-%m') AS month, COUNT(*) AS bills, DATEDIFF(CURDATE(),MAX(sale_date)) AS days_since_last
FROM sales GROUP BY month;

10. CASE – stock status label

SELECT p.name, SUM(s.quantity) AS qty,
 CASE WHEN SUM(s.quantity)=0 THEN 'Out of Stock'
      WHEN SUM(s.quantity)<=p.reorder_level THEN 'Low Stock' ELSE 'OK' END AS status
FROM products p JOIN stock s ON s.product_id=p.product_id GROUP BY p.product_id, p.name, p.reorder_level;

JOIN Operations (10)

1. INNER JOIN – product with category

Lists each product with its category name; 11 rows.

SELECT p.name, c.name AS category FROM products p INNER JOIN categories c ON p.category_id=c.category_id;

2. INNER JOIN – product with supplier

Shows who supplies each product.

SELECT p.name, s.name AS supplier, s.city FROM products p JOIN suppliers s ON p.supplier_id=s.supplier_id;

3. LEFT JOIN – products never sold

Expected: Extension Board, Water Bottle.

SELECT p.name FROM products p LEFT JOIN sales_items si ON p.product_id=si.product_id WHERE si.item_id IS NULL;

4. RIGHT JOIN – warehouses with or without stock rows

Every warehouse appears, even if empty.

SELECT w.name, COUNT(s.product_id) AS items FROM stock s RIGHT JOIN warehouses w ON s.warehouse_id=w.warehouse_id GROUP BY w.name;

5. SELF JOIN – products of the same category with higher price

Pairs a cheaper product with a costlier one in its category.

SELECT a.name AS cheaper, b.name AS costlier FROM products a JOIN products b
 ON a.category_id=b.category_id AND a.unit_price<b.unit_price;

6. Multi-table – purchase order details

PO, supplier, product, quantity and line cost.

SELECT po.po_id, s.name AS supplier, p.name AS product, pi.quantity, pi.quantity*pi.unit_cost AS line_cost
FROM purchase_orders po JOIN suppliers s ON po.supplier_id=s.supplier_id
JOIN purchase_items pi ON pi.po_id=po.po_id JOIN products p ON p.product_id=pi.product_id;

7. Multi-table – sales invoice lines

Customer, warehouse, product and amount per sale line.

SELECT sa.sale_id, sa.customer_name, w.name AS warehouse, p.name, si.quantity*si.unit_price AS amount
FROM sales sa JOIN warehouses w USING(warehouse_id) JOIN sales_items si USING(sale_id) JOIN products p USING(product_id);

8. Stock by warehouse

Where each product is physically held.

SELECT w.name AS warehouse, p.name AS product, s.quantity FROM stock s
JOIN products p ON p.product_id=s.product_id JOIN warehouses w ON w.warehouse_id=s.warehouse_id ORDER BY w.name;

9. LEFT JOIN – suppliers with no purchase orders

Identifies idle suppliers.

SELECT s.name FROM suppliers s LEFT JOIN purchase_orders po ON s.supplier_id=po.supplier_id WHERE po.po_id IS NULL;

10. Stock movement history

Audit trail of IN/OUT movement with product and warehouse names.

SELECT t.txn_date, p.name, w.name AS warehouse, t.txn_type, t.quantity, t.reference
FROM stock_transactions t JOIN products p USING(product_id) JOIN warehouses w USING(warehouse_id) ORDER BY t.txn_date;

Subqueries (5)

1. Products that have never been ordered by customers

SELECT name FROM products WHERE product_id NOT IN (SELECT product_id FROM sales_items);

2. Products priced above the average price

SELECT name, unit_price FROM products WHERE unit_price > (SELECT AVG(unit_price) FROM products);

3. Second-highest priced product

SELECT name, unit_price FROM products WHERE unit_price = (SELECT MAX(unit_price) FROM products WHERE unit_price < (SELECT MAX(unit_price) FROM products));

4. Customers whose bill is above the average bill

SELECT sale_id, customer_name, SUM(quantity*unit_price) AS bill FROM sales JOIN sales_items USING(sale_id) GROUP BY sale_id, customer_name
HAVING bill > (SELECT AVG(t) FROM (SELECT SUM(quantity*unit_price) t FROM sales_items GROUP BY sale_id) x);

5. Products whose total stock is below reorder level

SELECT name FROM products p WHERE (SELECT COALESCE(SUM(quantity),0) FROM stock s WHERE s.product_id=p.product_id) <= p.reorder_level;

Business Reports

1. Current Stock Report

SELECT p.sku, p.name, c.name AS category, SUM(s.quantity) AS total_stock
FROM products p JOIN categories c USING(category_id) LEFT JOIN stock s USING(product_id) GROUP BY p.product_id, p.sku, p.name, c.name;

2. Low-Stock Alert Report

SELECT * FROM v_low_stock;   -- view defined in Advanced SQL

3. Supplier-wise Purchase Report

SELECT s.name AS supplier, COUNT(DISTINCT po.po_id) AS orders, SUM(pi.quantity*pi.unit_cost) AS purchase_value
FROM suppliers s JOIN purchase_orders po USING(supplier_id) JOIN purchase_items pi USING(po_id) GROUP BY s.supplier_id, s.name ORDER BY purchase_value DESC;

4. Monthly Sales Report

SELECT DATE_FORMAT(sale_date,'%Y-%m') AS month, COUNT(DISTINCT sale_id) AS bills, SUM(quantity*unit_price) AS revenue
FROM sales JOIN sales_items USING(sale_id) GROUP BY month ORDER BY month;

5. Fast-Moving Products (top sellers)

SELECT p.name, SUM(si.quantity) AS units_sold FROM sales_items si JOIN products p USING(product_id)
GROUP BY p.product_id, p.name ORDER BY units_sold DESC LIMIT 5;

6. Slow-Moving Products (few or no sales)

SELECT p.name, COALESCE(SUM(si.quantity),0) AS units_sold FROM products p LEFT JOIN sales_items si USING(product_id)
GROUP BY p.product_id, p.name HAVING units_sold < 5 ORDER BY units_sold;

7. Pending Purchase Orders

SELECT po.po_id, s.name, po.order_date FROM purchase_orders po JOIN suppliers s USING(supplier_id) WHERE po.status='Pending';

Advanced SQL

Views

CREATE VIEW v_low_stock AS
SELECT p.product_id, p.name, p.reorder_level, COALESCE(SUM(s.quantity),0) AS total_stock
FROM products p LEFT JOIN stock s USING(product_id) GROUP BY p.product_id, p.name, p.reorder_level
HAVING total_stock <= p.reorder_level;

CREATE VIEW v_monthly_sales AS
SELECT DATE_FORMAT(sale_date,'%Y-%m') AS month, SUM(quantity*unit_price) AS revenue
FROM sales JOIN sales_items USING(sale_id) GROUP BY month;

Transactions

Transfer 5 Notebooks from Chennai to Main Warehouse atomically; ROLLBACK if anything fails.

START TRANSACTION;
UPDATE stock SET quantity = quantity-5 WHERE product_id=8 AND warehouse_id=3;
INSERT INTO stock VALUES (8,1,5) ON DUPLICATE KEY UPDATE quantity = quantity+5;
COMMIT;

START TRANSACTION;
DELETE FROM stock WHERE product_id=8;
ROLLBACK;   -- undo: stock rows restored

Stored Procedures

DELIMITER //
CREATE PROCEDURE sp_receive_po(IN p_po INT, IN p_wh INT)
BEGIN
 START TRANSACTION;
 INSERT INTO stock(product_id,warehouse_id,quantity)
  SELECT product_id,p_wh,quantity FROM purchase_items WHERE po_id=p_po
  ON DUPLICATE KEY UPDATE quantity = stock.quantity + VALUES(quantity);
 INSERT INTO stock_transactions(product_id,warehouse_id,txn_type,quantity,txn_date,reference)
  SELECT product_id,p_wh,'IN',quantity,CURDATE(),CONCAT('PO-',p_po) FROM purchase_items WHERE po_id=p_po;
 UPDATE purchase_orders SET status='Received' WHERE po_id=p_po;
 COMMIT;
END//

CREATE PROCEDURE sp_sell_item(IN p_sale INT, IN p_prod INT, IN p_qty INT, IN p_wh INT)
BEGIN
 DECLARE avail INT;
 SELECT quantity INTO avail FROM stock WHERE product_id=p_prod AND warehouse_id=p_wh;
 IF avail IS NULL OR avail < p_qty THEN
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Insufficient stock';
 ELSE
  INSERT INTO sales_items(sale_id,product_id,quantity,unit_price)
   SELECT p_sale,p_prod,p_qty,unit_price FROM products WHERE product_id=p_prod;
 END IF;
END//
DELIMITER ;
CALL sp_receive_po(4,1);

Trigger

Automatically reduces stock and logs an OUT transaction whenever a sale line is added.

DELIMITER //
CREATE TRIGGER trg_sale_item_ai AFTER INSERT ON sales_items FOR EACH ROW
BEGIN
 DECLARE wh INT;
 SELECT warehouse_id INTO wh FROM sales WHERE sale_id=NEW.sale_id;
 UPDATE stock SET quantity = quantity-NEW.quantity WHERE product_id=NEW.product_id AND warehouse_id=wh;
 INSERT INTO stock_transactions(product_id,warehouse_id,txn_type,quantity,txn_date,reference)
 VALUES (NEW.product_id,wh,'OUT',NEW.quantity,CURDATE(),CONCAT('SALE-',NEW.sale_id));
END//
DELIMITER ;

If you load the sample data above with this trigger already created, stock will be reduced twice – create the trigger after loading the data.

Index and Performance Comparison

EXPLAIN SELECT * FROM sales_items WHERE product_id=8;   -- type=ALL (full scan)
CREATE INDEX idx_sales_items_product ON sales_items(product_id);
EXPLAIN SELECT * FROM sales_items WHERE product_id=8;   -- type=ref, rows scanned drop
CREATE INDEX idx_sales_date ON sales(sale_date);        -- speeds monthly reports

With a small sample the difference is tiny, but on thousands of rows the index replaces a full table scan with a direct lookup.
